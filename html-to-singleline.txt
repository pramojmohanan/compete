    <script>
        // Auto-convert on page load
        window.addEventListener('load', () => {
            convertToSingleLine();
        });

        // Auto-convert on input change
        document.getElementById('input').addEventListener('input', convertToSingleLine);

        // Auto-convert on checkbox change
        document.querySelectorAll('input[type="checkbox"]').forEach(checkbox => {
            checkbox.addEventListener('change', convertToSingleLine);
        });

        function convertToSingleLine() {
            const input = document.getElementById('input').value;
            const removeComments = document.getElementById('removeComments').checked;
            const removeExtraSpaces = document.getElementById('removeExtraSpaces').checked;
            const minifyAttributes = document.getElementById('minifyAttributes').checked;

            let result = input;

            // Remove HTML comments
            if (removeComments) {
                result = result.replace(/<!--[\s\S]*?-->/g, '');
            }

            // Remove newlines and tabs
            result = result.replace(/[\r\n\t]/g, ' ');

            // Remove extra spaces
            if (removeExtraSpaces) {
                // Remove spaces between tags
                result = result.replace(/>\s+</g, '><');
                
                // Remove multiple spaces
                result = result.replace(/\s{2,}/g, ' ');
                
                // Trim spaces around equals in attributes
                if (minifyAttributes) {
                    result = result.replace(/\s*=\s*/g, '=');
                }
            }

            // Trim the result
            result = result.trim();

            // Update output
            document.getElementById('output').value = result;

            // Update stats
            updateStats(input, result);
        }

        function updateStats(original, converted) {
            const originalSize = original.length;
            const convertedSize = converted.length;
            const reduction = originalSize > 0 
                ? ((originalSize - convertedSize) / originalSize * 100).toFixed(1)
                : 0;

            document.getElementById('originalSize').textContent = originalSize;
            document.getElementById('convertedSize').textContent = convertedSize;
            document.getElementById('reduction').textContent = reduction + '%';
        }

        function copyOutput() {
            const output = document.getElementById('output');
            output.select();
            document.execCommand('copy');

            // Visual feedback
            const btn = event.target;
            const originalText = btn.textContent;
            btn.textContent = 'Copied!';
            btn.classList.add('copied');

            setTimeout(() => {
                btn.textContent = originalText;
                btn.classList.remove('copied');
            }, 2000);
        }
    </script>
